declare global {

      user?: {
        id: string;
        email: string;
        name: string;

      sessionId?: string;
    }
  }
}

const SESSION_COOKIE = "session_id";
const SESSION_DURATION_MS = 7 * 24 * 60 * 60 * 1000;

async function authMiddleware(req: Request, res: Response, next: NextFunction) {
  const sessionId = req.cookies?.[SESSION_COOKIE];

  if (!sessionId) {
    return next();
  }

  try {
    const session = await storage.getSession(sessionId);
    if (session) {
      const user = await storage.getUser(session.userId);
      if (user) {
        req.user = {
          id: user.id,
          email: user.email,
          name: user.name,
        };
        req.sessionId = sessionId;
      }
    }
  } catch (error) {
    console.error("Auth middleware error:", error);
  }

  next();
}

  app.use(authMiddleware);

    try {
      const result = registerSchema.safeParse(req.body);

      if (!result.success) {
        const validationError = fromError(result.error);
        return res.status(400).json({
          error: "Validation failed",
          details: validationError.toString()
        });
      }

      const { email, password, name } = result.data;

      const existingUser = await storage.getUserByEmail(email);
      if (existingUser) {
        return res.status(400).json({ error: "Email already registered" });
      }

      const hashedPassword = await bcrypt.hash(password, 10);
      const user = await storage.createUser({
        email,
        password: hashedPassword,
        name,
      });

      const expiresAt = new Date(Date.now() + SESSION_DURATION_MS);
      const session = await storage.createSession(user.id, expiresAt);

      res.cookie(SESSION_COOKIE, session.id, {
        httpOnly: true,
        secure: process.env.NODE_ENV === "production",
        sameSite: "lax",
        maxAge: SESSION_DURATION_MS,
      });

      return res.status(201).json({
        user: { id: user.id, email: user.email, name: user.name },
      });
    } catch (error) {
      console.error("Error registering user:", error);

        httpOnly: true,
        secure: process.env.NODE_ENV === "production",
        sameSite: "lax",
        maxAge: SESSION_DURATION_MS,

async function generatePaymentUrl(order: { id: string; orderNumber: string; amount: string; currency: string; customerEmail: string; customerName: string }): Promise<{ paymentUrl: string; gatewayOrderId?: string }> {
  const baseUrl = process.env.PAYMENT_GATEWAY_URL;
  const identityApiUrl = process.env.IDENTITY_API_URL;
  const networkGatewayApiKey = process.env.NETWORK_PG_KEY;
  const redirectUrl = process.env.REDIRECT_URL;
  const cancelUrl = process.env.CANCEL_URL;
  console.log(baseUrl,identityApiUrl,networkGatewayApiKey);
  if (!baseUrl || !identityApiUrl || !networkGatewayApiKey) {
    const appUrl = process.env.REPL_SLUG
      ? `https://${process.env.REPL_SLUG}.${process.env.REPL_OWNER}.repl.co`
      : "http://localhost:5000";
    return { paymentUrl: `${appUrl}/payment?orderId=${order.id}&orderNumber=${order.orderNumber}` };
  }

  const { data } = await axios({
    method: "post",
    url: identityApiUrl,
    headers: {
      "Content-Type": "application/vnd.ni-identity.v1+json",
      Authorization: `Basic ${networkGatewayApiKey}`
    },
    data: {
      grant_type: "client_credentials"
    }
  });

  const { access_token } = data;

  const { data: orderData } = await axios({
    method: "post",
    url: baseUrl,
    headers: {
      Authorization: `Bearer ${access_token}`,
      "Content-Type": "application/vnd.ni-payment.v2+json",
      Accept: "application/vnd.ni-payment.v2+json",
    },
    data: {
      action: "PURCHASE",
      amount: {
        currencyCode: order.currency,
        value: Number(Number(order.amount) * 100)
      },
      merchantOrderReference: order.orderNumber,
      merchantAttributes: {
        redirectUrl: redirectUrl || `${process.env.APP_URL || ""}/payment/callback`,
        skipConfirmationPage: true,
        cancelUrl: cancelUrl || `${process.env.APP_URL || ""}/payment/cancelled`,
      }
    }
  });

  const paymentUrl = orderData?._links?.payment?.href || "";
  const gatewayOrderId = orderData?.reference || orderData?._id;

  return { paymentUrl, gatewayOrderId };
}
